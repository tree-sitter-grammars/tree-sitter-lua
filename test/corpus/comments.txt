--
-- single-line comment, below an empty single-line comment

--[[]]

--[[
print("block comment")
--]]

--[[
print("block comment")
]]

-- [[
print("not block comment")
--]]

--[=[[print("level 1 block comment")]]=]

--[=[
  [print("level 1 block comment")]
]=]

--[=[
  [print("level 1 block comment")
]=]

--[=[
  print("level 1 block comment")]
]=]

--[=[
  [[print("level 1 block comment")
]=]

--[=[
  print("level 1 block comment")]]
]=]

--[==[
print("level 2 block comment")
-- ]==]

--[==[
print("level 2 block comment")
]==]

-- [==[
print("not level 2 block comment")
--]==]

print "--[[hello]]world" -- long comment delimiters inside a string
print "--[=["
print "foo"
--]=]
print "bar"
print '---[==['           -- the same case in a single-quoted string
